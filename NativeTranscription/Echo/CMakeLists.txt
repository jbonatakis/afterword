cmake_minimum_required(VERSION 3.15)
project(RecorderEcho LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT DEFINED WEBRTC_SOURCE)
    message(FATAL_ERROR "WEBRTC_SOURCE must point to the pinned vendored WebRTC sources")
endif()
add_subdirectory("${WEBRTC_SOURCE}" webrtc)
add_library(RecorderEcho STATIC EchoStream.cpp)
target_include_directories(RecorderEcho PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(RecorderEcho PUBLIC webrtc_audio_static "-framework CoreFoundation")
add_executable(EchoTests ../../Tests/EchoTests.cpp)
target_link_libraries(EchoTests PRIVATE RecorderEcho)
